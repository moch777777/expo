// App.js
import React, { useState, useEffect } from 'react';
import { SafeAreaView, Text, TextInput, Button, FlatList, Alert } from 'react-native';

const SERVER = 'http://localhost:3000'; // später Cloud-Server

export default function App() {
  const [userId,setUserId] = useState(null);
  const [email,setEmail] = useState('');
  const [password,setPassword] = useState('');
  const [accounts,setAccounts] = useState([]);
  const [transactions,setTransactions] = useState([]);
  const [selectedAccount,setSelectedAccount] = useState(null);

  // Login
  const login = async()=>{
    try{
      const res = await fetch(`${SERVER}/login`,{method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({email,password})});
      const data = await res.json();
      if(data.success){
        setUserId(data.userId);
        fetchAccounts(data.userId);
      } else Alert.alert('Login fehlgeschlagen');
    }catch(e){ Alert.alert('Fehler', e.message);}
  };

  const fetchAccounts = async(uid)=>{
    const res = await fetch(`${SERVER}/accounts/${uid}`);
    const data = await res.json();
    setAccounts(data);
    if(data[0]) selectAccount(data[0]);
  }

  const selectAccount = async(account)=>{
    setSelectedAccount(account);
    const res = await fetch(`${SERVER}/transactions/${account.id}`);
    const txs = await res.json();
    setTransactions(txs);
  }

  const transferDemo = async()=>{
    const amount = 5; // Demo
    const res = await fetch(`${SERVER}/transfer`,{method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({fromAccountId:selectedAccount.id,toAccount:'demo',amount})});
    const data = await res.json();
    if(data.success){
      Alert.alert('Überweisung erfolgreich', `Neuer Kontostand: ${data.newBalance} EUR`);
      fetchAccounts(userId);
    }
  }

  if(!userId){
    return (
      <SafeAreaView style={{padding:20}}>
        <Text style={{fontSize:24}}>Revo24 Login</Text>
        <TextInput placeholder="Email" value={email} onChangeText={setEmail} style={{borderWidth:1,margin:5,padding:8}}/>
        <TextInput placeholder="Passwort" value={password} secureTextEntry onChangeText={setPassword} style={{borderWidth:1,margin:5,padding:8}}/>
        <Button title="Login" onPress={login}/>
      </SafeAreaView>
    )
  }

  return (
    <SafeAreaView style={{padding:20}}>
      <Text style={{fontSize:24}}>Willkommen bei Revo24!</Text>
      <Text style={{marginVertical:10}}>Konten:</Text>
      <FlatList
        data={accounts}
        keyExtractor={item=>item.id}
        renderItem={({item})=>(
          <Text style={{fontSize:18}} onPress={()=>selectAccount(item)}>{item.name}: {item.balance} {item.currency}</Text>
        )}
      />
      {selectedAccount &&
        <>
          <Text style={{marginTop:10, fontSize:20}}>Transaktionen für {selectedAccount.name}</Text>
          <FlatList
            data={transactions}
            keyExtractor={item=>item.id}
            renderItem={({item})=><Text>{item.date}: {item.amount} EUR - {item.description}</Text>}
          />
          <Button title="Demo-Überweisung 5€" onPress={transferDemo}/>
        </>
      }
      <Button title="Logout" onPress={()=>{setUserId(null); setAccounts([]); setTransactions([])}}/>
